1505    ///
1506    /// # Examples
1507    /// ```
1508    /// use bad64::SysReg;
1509    /// assert_eq!(SysReg::VBAR_EL3.name(), "vbar_el3");
1510    /// ```
1511    ///
1512    /// ```
1513    /// use bad64::{decode, Operand, SysReg};

1515    /// let decoded = decode(0xd51ec000, 0).unwrap();
1516    ///
1517    /// let op = decoded.operands()[0];
1518    ///
1519    /// assert_eq!(op, Operand::SysReg(SysReg::VBAR_EL3));
1520    ///
1521    /// match op {
1522    ///     Operand::SysReg(sr) => assert_eq!(sr.name(), "vbar_el3"),
1523    ///     _ => assert!(false),
1524    /// };
1525    /// ```
1526    pub fn name(&self) -> &'static str {
1527        #[cfg(target_os = "windows")]
1528        {
1529            unsafe {
1530                CStr::from_ptr(bad64_sys::get_system_register_name(self.to_i32().unwrap()) as _)
1531            }
1532            .to_str()
1533            .unwrap()
1534        }
1535        #[cfg(not(target_os = "windows"))]
1536        {
1537            unsafe {
1538                CStr::from_ptr(bad64_sys::get_system_register_name(self.to_u32().unwrap()) as _)
1539            }
1540            .to_str()
1541            .unwrap()
1542        }
1543    }
1544}
