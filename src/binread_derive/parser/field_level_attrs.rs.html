58impl StructField {
59    pub(crate) fn can_call_after_parse(&self) -> bool {
60        matches!(self.read_mode, ReadMode::Normal) && !self.map.is_some()
61    }
62
63    pub(crate) fn should_use_after_parse(&self) -> bool {
64        !*self.deref_now
65    }
66
67    pub(crate) fn generated_value(&self) -> bool {
68        matches!(self.read_mode, ReadMode::Calc(_) | ReadMode::Default)
69    }
70
71    fn validate(&self) -> syn::Result<()> {
72        if let (Some(offset_after), true) = (&self.offset_after, *self.deref_now) {
73            let offset_after_span = offset_after.span();
74            let span = offset_after_span
75                .join(self.deref_now.span())
76                .unwrap_or(offset_after_span);
77            Err(syn::Error::new(
78                span,
79                "`deref_now` and `offset_after` are mutually exclusive",
80            ))
81        } else {
82            Ok(())
83        }
84    }
85}

87impl FromField for StructField {
88    type In = syn::Field;
89
90    fn from_field(field: &Self::In, index: usize) -> ParseResult<Self> {
91        let result = Self::set_from_attrs(
92            Self {
93                ident: field
94                    .ident
95                    .clone()
96                    .unwrap_or_else(|| quote::format_ident!("self_{}", index)),
97                generated_ident: field.ident.is_none(),
98                ty: field.ty.clone(),
99                endian: <_>::default(),
100                map: <_>::default(),
101                magic: <_>::default(),
102                args: <_>::default(),
103                read_mode: <_>::default(),
104                count: <_>::default(),
105                offset: <_>::default(),
106                offset_after: <_>::default(),
107                if_cond: <_>::default(),
108                deref_now: <_>::default(),
109                restore_position: <_>::default(),
110                do_try: <_>::default(),
111                temp: <_>::default(),
112                assertions: <_>::default(),
113                pad_before: <_>::default(),
114                pad_after: <_>::default(),
115                align_before: <_>::default(),
116                align_after: <_>::default(),
117                seek_before: <_>::default(),
118                pad_size_to: <_>::default(),
119            },
120            &field.attrs,
121        );
122
123        match result {
124            ParseResult::Ok(this) => {
125                if let Err(error) = this.validate() {
126                    ParseResult::Partial(this, error)
127                } else {
128                    ParseResult::Ok(this)
129                }
130            }
131            ParseResult::Partial(this, mut parse_error) => {
132                if let Err(error) = this.validate() {
133                    parse_error.combine(error);
134                }
135                ParseResult::Partial(this, parse_error)
136            }
137            ParseResult::Err(error) => ParseResult::Err(error),
138        }
139    }
140}

169#[derive(Clone, Debug)]
170pub(crate) enum EnumVariant {
171    Variant { ident: syn::Ident, options: Struct },
172    Unit(UnitEnumField),
173}
174
175impl EnumVariant {
176    pub(crate) fn ident(&self) -> &syn::Ident {
177        match self {
178            EnumVariant::Variant { ident, .. } => &ident,
179            EnumVariant::Unit(field) => &field.ident,
180        }
181    }
182}
183
184impl FromField for EnumVariant {
185    type In = syn::Variant;
186
187    fn from_field(variant: &Self::In, index: usize) -> ParseResult<Self> {
188        match variant.fields {
189            syn::Fields::Named(_) | syn::Fields::Unnamed(_) => {
190                Struct::from_input(&variant.attrs, variant.fields.iter()).map(|options| {
191                    Self::Variant {
192                        ident: variant.ident.clone(),
193                        options,
194                    }
195                })
196            }
197            syn::Fields::Unit => UnitEnumField::from_field(variant, index).map(Self::Unit),
198        }
199    }
