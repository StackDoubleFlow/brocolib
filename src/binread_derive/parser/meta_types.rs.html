13/// `MetaExpr` represents a key/expr pair
14/// Takes two forms:
15/// * ident(expr)
16/// * ident = expr
17/// both are always allowed
18pub(crate) type MetaExpr<Keyword> = MetaValue<Keyword, Expr>;

20/// `MetaType` represents a key/ty pair
21/// Takes two forms:
22/// * ident(ty)
23/// * ident = ty
24/// both are always allowed
25pub(crate) type MetaType<Keyword> = MetaValue<Keyword, Type>;

27/// `MetaLit` represents a key/lit pair
28/// Takes two forms:
29/// * ident(lit)
30/// * ident = lit
31/// both are always allowed
32pub(crate) type MetaLit<Keyword> = MetaValue<Keyword, Lit>;

34#[derive(Debug, Clone)]
35pub(crate) struct MetaValue<Keyword, Value> {
36    pub(crate) ident: Keyword,
37    pub(crate) value: Value,
38}
39
40impl<Keyword: Parse, Value: Parse> Parse for MetaValue<Keyword, Value> {
41    fn parse(input: ParseStream<'_>) -> syn::Result<Self> {
42        let ident = input.parse()?;
43        let value = if input.peek(token::Paren) {
44            let content;
45            parenthesized!(content in input);
46            content.parse()?
47        } else {
48            input.parse::<Token![=]>()?;
49            input.parse()?
50        };
51
52        Ok(MetaValue { ident, value })
53    }
54}

56impl<Keyword, Value: ToTokens> From<MetaValue<Keyword, Value>> for TokenStream {

77pub(crate) struct MetaList<Keyword, ItemType> {
78    pub(crate) ident: Keyword,
79    pub(crate) parens: token::Paren,
80    pub(crate) fields: Fields<ItemType>,
81}

83impl<Keyword: Parse, ItemType: Parse> Parse for MetaList<Keyword, ItemType> {
84    fn parse(input: ParseStream<'_>) -> syn::Result<Self> {
85        let ident = input.parse()?;
86        let content;
87        let parens = parenthesized!(content in input);
88        Ok(MetaList {
89            ident,
90            parens,
91            fields: content.parse_terminated::<_, Token![,]>(ItemType::parse)?,
92        })
93    }
94}

106// This is like `syn::PatType` except:
107// (1) Implements `Parse`;
108// (2) No attributes;
109// (3) Only allows an ident on the LHS instead of any `syn::Pat`.
110#[derive(Debug, Clone)]
111pub(crate) struct IdentPatType {
112    pub(crate) ident: syn::Ident,
113    pub(crate) colon_token: Token![:],
114    pub(crate) ty: syn::Type,
115}
116
117impl Parse for IdentPatType {
118    fn parse(input: ParseStream<'_>) -> syn::Result<Self> {
119        Ok(IdentPatType {
120            ident: input.parse()?,
121            colon_token: input.parse()?,
122            ty: input.parse()?,
123        })
124    }
125}

127pub(crate) struct MetaAttrList<P>(Fields<P>);

129impl<P> MetaAttrList<P> {
130    pub(crate) fn into_iter(self) -> impl Iterator<Item = P> {

135impl<P: Parse> Parse for MetaAttrList<P> {
136    fn parse(input: ParseStream<'_>) -> syn::Result<Self> {
137        let content;
138        parenthesized!(content in input);
139        Ok(MetaAttrList(Fields::parse_terminated(&content)?))
140    }
