12pub(super) fn generate_unit_enum(input: &Input, en: &UnitOnlyEnum) -> TokenStream {
13    match &en.repr {
14        Some(repr) => generate_unit_enum_repr(input, repr, &en.fields),
15        None => generate_unit_enum_magic(input, en, &en.fields),
16    }
17}

19fn generate_unit_enum_repr(
20    input: &Input,
21    repr: &TokenStream,
22    variants: &[UnitEnumField],
23) -> TokenStream {
24    let clauses = variants.iter().map(|variant| {
25        let ident = &variant.ident;
26        quote! {
27            if #TEMP == Self::#ident as #repr {
28                Ok(Self::#ident)
29            }
30        }
31    });
32
33    let prelude = PreludeGenerator::new(input).add_options().finish();
34
35    quote! {
36        #prelude
37        let #TEMP: #repr = #READ_METHOD(#READER, #OPT, ())?;
38        #(#clauses else)* {
39            Err(#BIN_ERROR::NoVariantMatch {
40                pos: #POS,
41            })
42        }
43    }
44}

46fn generate_unit_enum_magic(
47    input: &Input,
48    en: &UnitOnlyEnum,
49    variants: &[UnitEnumField],
50) -> TokenStream {
51    let prelude = PreludeGenerator::new(input)
52        .add_imports()
53        .add_options()
54        .finish();
55
56    let matches = variants.iter().filter_map(|field| {
57        if let Some(magic) = &field.magic {
58            let ident = &field.ident;
59            let magic = magic.match_value();
60            let condition = if field.pre_assertions.is_empty() {
61                quote! { #magic }
62            } else {
63                let pre_assertions = field.pre_assertions.iter().map(|assert| &assert.condition);
64                quote! { #magic if true #(&& (#pre_assertions))* }
65            };
66            Some(quote! { #condition => Ok(Self::#ident) })
67        } else {
68            None
69        }
70    });
71
72    let amp = en
73        .expected_field_magic
74        .as_ref()
75        .map(|magic| magic.add_ref());
76
77    quote! {
78        #prelude
79        match #amp#READ_METHOD(#READER, #OPT, ())? {
80            #(#matches,)*
81            _ => Err(#BIN_ERROR::NoVariantMatch { pos: #POS })
82        }
83    }
84}

86pub(super) fn generate_data_enum(en: &Enum) -> TokenStream {
87    let return_all_errors = en.error_mode != EnumErrorMode::ReturnUnexpectedError;
88
89    let (create_error_basket, return_error) = if return_all_errors {
90        (
91            quote! {
92                extern crate alloc;
93                let mut #ERROR_BASKET: alloc::vec::Vec<(&'static str, #BIN_ERROR)> = alloc::vec::Vec::new();
94            },
95            quote! {
96                Err(#BIN_ERROR::EnumErrors {
97                    pos: #POS,
98                    variant_errors: #ERROR_BASKET
99                })
100            },
101        )
102    } else {
103        (
104            TokenStream::new(),
105            quote! {
106                Err(#BIN_ERROR::NoVariantMatch {
107                    pos: #POS
108                })
109            },
110        )
111    };
112
113    let try_each_variant = en.variants.iter().map(|variant| {
114        let body = generate_variant_impl(en, variant);
115
116        let handle_error = if return_all_errors {
117            let name = variant.ident().to_string();
118            quote! {
119                #ERROR_BASKET.push((#name, #TEMP.err().unwrap()));
120            }
121        } else {
122            TokenStream::new()
123        };
124
125        quote! {
126            let #TEMP = (|| {
127                #body
128            })();
129
130            if #TEMP.is_ok() {
131                return #TEMP;
132            } else {
133                #handle_error
134                #SEEK_TRAIT::seek(#READER, #SEEK_FROM::Start(#POS))?;
135            }
136        }
137    });
138
139    quote! {
140        #create_error_basket
141        #(#try_each_variant)*
142        #return_error
143    }
144}

146fn generate_variant_impl(en: &Enum, variant: &EnumVariant) -> TokenStream {
147    // TODO: Kind of expensive since the enum is containing all the fields
148    // and this is a clone.
149    let input = Input::Enum(en.with_variant(variant));
150
151    match variant {
152        EnumVariant::Variant { ident, options } => StructGenerator::new(&input, &options)
153            .read_fields()
154            .add_assertions(get_assertions(&en.assertions))
155            .return_value(Some(ident))
156            .finish(),
157
158        EnumVariant::Unit(options) => generate_unit_struct(&input, Some(&options.ident)),
159    }
160}
