4use crate::formatter::fast::FastStringRegister;
5use crate::formatter::regs_tbl::*;
6use crate::iced_constants::IcedConstants;
7use alloc::boxed::Box;
8use alloc::vec::Vec;
9use core::convert::TryInto;
10use lazy_static::lazy_static;
11
12lazy_static! {
13	pub(super) static ref REGS_TBL: Box<[FastStringRegister; IcedConstants::REGISTER_ENUM_COUNT]> = create();
14}
15
16fn create() -> Box<[FastStringRegister; IcedConstants::REGISTER_ENUM_COUNT]> {
17	// If this fails, the generator was updated and now FastStringRegister must be changed
18	// to the correct type in fast.rs
19	const _: () = assert!(FastStringRegister::SIZE == VALID_STRING_LENGTH);
20
21	let mut result = Vec::with_capacity(IcedConstants::REGISTER_ENUM_COUNT);
22	let mut data = &REGS_DATA[..];
23	for _ in 0..IcedConstants::REGISTER_ENUM_COUNT {
24		let len = data[0] as usize;
25
26		// It's safe to read FastStringRegister::SIZE bytes from the last string since the
27		// table includes extra padding. See const-assert above and the table.
28		result.push(FastStringRegister::new(data.as_ptr()));
29
30		data = &data[1 + len..];
31	}
32	debug_assert!(data.len() == PADDING_SIZE);
33
34	#[allow(clippy::unwrap_used)]
35	result.into_boxed_slice().try_into().ok().unwrap()
36}
