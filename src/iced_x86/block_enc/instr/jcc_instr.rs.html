10#[derive(Debug, Copy, Clone, Eq, PartialEq, Ord, PartialOrd)]
11enum InstrKind {
12	Unchanged,
13	Short,
14	Near,
15	Long,
16	Uninitialized,
17}
18
19pub(super) struct JccInstr {
20	bitness: u8,
21	instruction: Instruction,
22	target_instr: TargetInstr,
23	pointer_data: Option<Rc<RefCell<BlockData>>>,
24	instr_kind: InstrKind,
25	short_instruction_size: u8,
26	near_instruction_size: u8,
27	long_instruction_size64: u8,
28}

30impl JccInstr {
31	#[inline]
32	#[allow(unused_variables)]
33	fn long_instruction_size64(instruction: &Instruction) -> u32 {
34		#[cfg(feature = "mvex")]
35		{
36			// Check if JKZD/JKNZD
37			if instruction.op_count() == 2 {
38				return 5 + InstrUtils::CALL_OR_JMP_POINTER_DATA_INSTRUCTION_SIZE64;
39			}
40		}
41		// Code:
42		//		!jcc short skip		; negated jcc opcode
43		//		jmp qword ptr [rip+mem]
44		//	skip:
45		2 + InstrUtils::CALL_OR_JMP_POINTER_DATA_INSTRUCTION_SIZE64
46	}
47
48	pub(super) fn new(block_encoder: &mut BlockEncInt, base: &mut InstrBase, instruction: &Instruction) -> Self {
49		let mut instr_kind = InstrKind::Uninitialized;
50		let mut instr_copy: Instruction;
51		let short_instruction_size;
52		let near_instruction_size;
53		let long_instruction_size64 = Self::long_instruction_size64(instruction) as u8;
54		if !block_encoder.fix_branches() {
55			instr_kind = InstrKind::Unchanged;
56			instr_copy = *instruction;
57			instr_copy.set_near_branch64(0);
58			base.size = block_encoder.get_instruction_size(&instr_copy, 0);
59			short_instruction_size = 0;
60			near_instruction_size = 0;
61		} else {
62			instr_copy = *instruction;
63			instr_copy.set_code(instruction.code().as_short_branch());
64			instr_copy.set_near_branch64(0);
65			short_instruction_size = block_encoder.get_instruction_size(&instr_copy, 0) as u8;
66
67			instr_copy = *instruction;
68			instr_copy.set_code(instruction.code().as_near_branch());
69			instr_copy.set_near_branch64(0);
70			near_instruction_size = block_encoder.get_instruction_size(&instr_copy, 0) as u8;
71
72			base.size = if block_encoder.bitness() == 64 {
73				// Make sure it's not shorter than the real instruction. It can happen if there are extra prefixes.
74				cmp::max(near_instruction_size, long_instruction_size64)
75			} else {
76				near_instruction_size
77			} as u32;
78		}
79		Self {
80			bitness: block_encoder.bitness() as u8,
81			instruction: *instruction,
82			target_instr: TargetInstr::default(),
83			pointer_data: None,
84			instr_kind,
85			short_instruction_size,
86			near_instruction_size,
87			long_instruction_size64,
88		}
89	}

91	fn try_optimize(&mut self, base: &mut InstrBase, ctx: &mut InstrContext<'_>, gained: u64) -> bool {
92		if self.instr_kind == InstrKind::Unchanged || self.instr_kind == InstrKind::Short {
93			base.done = true;
94			return false;
95		}
96
97		let mut target_address = self.target_instr.address(ctx);
98		let mut next_rip = ctx.ip.wrapping_add(self.short_instruction_size as u64);
99		let mut diff = target_address.wrapping_sub(next_rip) as i64;
100		diff = InstrUtils::convert_diff_to_bitness_diff(self.bitness, correct_diff(self.target_instr.is_in_block(ctx.block), diff, gained));
101		if i8::MIN as i64 <= diff && diff <= i8::MAX as i64 {
102			if let Some(ref pointer_data) = self.pointer_data {
103				pointer_data.borrow_mut().is_valid = false;
104			}
105			self.instr_kind = InstrKind::Short;
106			base.size = self.short_instruction_size as u32;
107			base.done = true;
108			return true;
109		}
110
111		// If it's in the same block, we assume the target is at most 2GB away.
112		let mut use_near = self.bitness != 64 || self.target_instr.is_in_block(ctx.block);
113		if !use_near {
114			target_address = self.target_instr.address(ctx);
115			next_rip = ctx.ip.wrapping_add(self.near_instruction_size as u64);
116			diff = target_address.wrapping_sub(next_rip) as i64;
117			diff = InstrUtils::convert_diff_to_bitness_diff(self.bitness, correct_diff(self.target_instr.is_in_block(ctx.block), diff, gained));
118			use_near = i32::MIN as i64 <= diff && diff <= i32::MAX as i64;
119		}
120		if use_near {
121			if let Some(ref pointer_data) = self.pointer_data {
122				pointer_data.borrow_mut().is_valid = false;
123			}
124			if diff < (IcedConstants::MAX_INSTRUCTION_LENGTH as i64) * (i8::MIN as i64)
125				|| diff > (IcedConstants::MAX_INSTRUCTION_LENGTH as i64) * (i8::MAX as i64)
126			{
127				base.done = true;
128			}
129			self.instr_kind = InstrKind::Near;
130			base.size = self.near_instruction_size as u32;
131			return true;
132		}
133
134		if self.pointer_data.is_none() {
135			self.pointer_data = Some(ctx.block.alloc_pointer_location());
136		}
137		self.instr_kind = InstrKind::Long;
138		false
139	}

141	fn short_br_to_native_br(code: Code, bitness: u32) -> Code {
142		let (c16, c32, c64) = match code {
143			Code::Jo_rel8_16 | Code::Jo_rel8_32 | Code::Jo_rel8_64 => (Code::Jo_rel8_16, Code::Jo_rel8_32, Code::Jo_rel8_64),
144			Code::Jno_rel8_16 | Code::Jno_rel8_32 | Code::Jno_rel8_64 => (Code::Jno_rel8_16, Code::Jno_rel8_32, Code::Jno_rel8_64),
145			Code::Jb_rel8_16 | Code::Jb_rel8_32 | Code::Jb_rel8_64 => (Code::Jb_rel8_16, Code::Jb_rel8_32, Code::Jb_rel8_64),
146			Code::Jae_rel8_16 | Code::Jae_rel8_32 | Code::Jae_rel8_64 => (Code::Jae_rel8_16, Code::Jae_rel8_32, Code::Jae_rel8_64),
147			Code::Je_rel8_16 | Code::Je_rel8_32 | Code::Je_rel8_64 => (Code::Je_rel8_16, Code::Je_rel8_32, Code::Je_rel8_64),
148			Code::Jne_rel8_16 | Code::Jne_rel8_32 | Code::Jne_rel8_64 => (Code::Jne_rel8_16, Code::Jne_rel8_32, Code::Jne_rel8_64),
149			Code::Jbe_rel8_16 | Code::Jbe_rel8_32 | Code::Jbe_rel8_64 => (Code::Jbe_rel8_16, Code::Jbe_rel8_32, Code::Jbe_rel8_64),
150			Code::Ja_rel8_16 | Code::Ja_rel8_32 | Code::Ja_rel8_64 => (Code::Ja_rel8_16, Code::Ja_rel8_32, Code::Ja_rel8_64),
151			Code::Js_rel8_16 | Code::Js_rel8_32 | Code::Js_rel8_64 => (Code::Js_rel8_16, Code::Js_rel8_32, Code::Js_rel8_64),
152			Code::Jns_rel8_16 | Code::Jns_rel8_32 | Code::Jns_rel8_64 => (Code::Jns_rel8_16, Code::Jns_rel8_32, Code::Jns_rel8_64),
153			Code::Jp_rel8_16 | Code::Jp_rel8_32 | Code::Jp_rel8_64 => (Code::Jp_rel8_16, Code::Jp_rel8_32, Code::Jp_rel8_64),
154			Code::Jnp_rel8_16 | Code::Jnp_rel8_32 | Code::Jnp_rel8_64 => (Code::Jnp_rel8_16, Code::Jnp_rel8_32, Code::Jnp_rel8_64),
155			Code::Jl_rel8_16 | Code::Jl_rel8_32 | Code::Jl_rel8_64 => (Code::Jl_rel8_16, Code::Jl_rel8_32, Code::Jl_rel8_64),
156			Code::Jge_rel8_16 | Code::Jge_rel8_32 | Code::Jge_rel8_64 => (Code::Jge_rel8_16, Code::Jge_rel8_32, Code::Jge_rel8_64),
157			Code::Jle_rel8_16 | Code::Jle_rel8_32 | Code::Jle_rel8_64 => (Code::Jle_rel8_16, Code::Jle_rel8_32, Code::Jle_rel8_64),
158			Code::Jg_rel8_16 | Code::Jg_rel8_32 | Code::Jg_rel8_64 => (Code::Jg_rel8_16, Code::Jg_rel8_32, Code::Jg_rel8_64),
159			_ => {
160				#[cfg(feature = "mvex")]
161				{
162					if bitness == 64 {
163						match code {
164							Code::VEX_KNC_Jkzd_kr_rel8_64 | Code::VEX_KNC_Jknzd_kr_rel8_64 => return code,
165							_ => {}
166						}
167					}
168				}
169				unreachable!()
170			}
171		};
172
173		match bitness {
174			16 => c16,
175			32 => c32,
176			64 => c64,
177			_ => unreachable!(),
178		}
179	}
180}

182impl Instr for JccInstr {
183	fn get_target_instr(&mut self) -> (&mut TargetInstr, u64) {
184		(&mut self.target_instr, self.instruction.near_branch_target())
185	}
186
187	fn optimize(&mut self, base: &mut InstrBase, ctx: &mut InstrContext<'_>, gained: u64) -> bool {
188		self.try_optimize(base, ctx, gained)
189	}
190
191	fn encode(&mut self, base: &mut InstrBase, ctx: &mut InstrContext<'_>) -> Result<(ConstantOffsets, bool), IcedError> {
192		match self.instr_kind {
193			InstrKind::Unchanged | InstrKind::Short | InstrKind::Near => {
194				if self.instr_kind == InstrKind::Unchanged {
195					// nothing
196				} else if self.instr_kind == InstrKind::Short {
197					self.instruction.set_code(self.instruction.code().as_short_branch());
198				} else {
199					debug_assert!(self.instr_kind == InstrKind::Near);
200					self.instruction.set_code(self.instruction.code().as_near_branch());
201				}
202				self.instruction.set_near_branch64(self.target_instr.address(ctx));
203				ctx.block.encoder.encode(&self.instruction, ctx.ip).map_or_else(
204					|err| Err(IcedError::with_string(InstrUtils::create_error_message(err, &self.instruction))),
205					|_| Ok((ctx.block.encoder.get_constant_offsets(), true)),
206				)
207			}
208
209			InstrKind::Long => {
210				debug_assert!(self.pointer_data.is_some());
211				let pointer_data = self.pointer_data.clone().ok_or_else(|| IcedError::new("Internal error"))?;
212				pointer_data.borrow_mut().data = self.target_instr.address(ctx);
213				let mut instr = Instruction::default();
214				instr.set_code(Self::short_br_to_native_br(
215					self.instruction.code().negate_condition_code().as_short_branch(),
216					ctx.block.encoder.bitness(),
217				));
218				if self.instruction.op_count() == 1 {
219					instr.set_op0_kind(OpKind::NearBranch64);
220				} else {
221					#[cfg(feature = "mvex")]
222					{
223						debug_assert_eq!(self.instruction.op_count(), 2);
224						instr.set_op0_kind(OpKind::Register);
225						instr.set_op0_register(self.instruction.op0_register());
226						instr.set_op1_kind(OpKind::NearBranch64);
227					}
228					#[cfg(not(feature = "mvex"))]
229					unreachable!();
230				}
231				debug_assert!(ctx.block.encoder.bitness() == 64);
232				debug_assert!(self.long_instruction_size64 <= i8::MAX as u8);
233				instr.set_near_branch64(ctx.ip.wrapping_add(self.long_instruction_size64 as u64));
234				let instr_len = ctx
235					.block
236					.encoder
237					.encode(&instr, ctx.ip)
238					.map_err(|err| IcedError::with_string(InstrUtils::create_error_message(err, &self.instruction)))? as u32;
239				InstrUtils::encode_branch_to_pointer_data(
240					ctx.block,
241					false,
242					ctx.ip.wrapping_add(instr_len as u64),
243					pointer_data,
244					base.size - instr_len,
245				)
246				.map_or_else(
247					|err| Err(IcedError::with_string(InstrUtils::create_error_message(err, &self.instruction))),
248					|_| Ok((ConstantOffsets::default(), false)),
249				)
250			}
251
252			InstrKind::Uninitialized => unreachable!(),
253		}
254	}
