7/// Returns address to (g_CodegenRegistration, g_MetadataRegistration)
8pub fn find_registration(obj: &ObjectReader) -> loader::Result<(u64, u64)> {
9    // il2cpp_init -> Runtime::Init -> s_Il2CppCodegenRegistration
10    // From there, we read the arguments being passed into il2cpp_codegen_register
11    let il2cpp_init = obj
12        .find_export("il2cpp_init")?
13        .ok_or(Il2CppBinaryError::MissingIl2CppInit)?;
14
15    let runtime_init = nth_call(obj, il2cpp_init, 2)? as usize;
16    let code_registration = nth_indirect_call(
17        obj,
18        runtime_init as u64,
19        if obj.format() == object::BinaryFormat::Pe {
20            // On windows, there are extra calls to Thread::GetCurrentThreadId and SystemFutex::Wait in il2cpp_baselib.
21            // These are not present in linux as they are instead standard syscalls.
22            3
23        } else {
24            1
25        },
26    )?
27    .ok_or(Il2CppBinaryError::MissingRegistration)?;
28
29    // Collect the arguments to il2cpp_codegen_register
30    let il2cpp_codegen_register_call =
31        nth_matching(obj, code_registration, 1, |t| t.is_jmp_short_or_near())?;
32    let start_offset = obj.vaddr_conv(code_registration)? as usize;
33    let call_instr_offset = obj.vaddr_conv(il2cpp_codegen_register_call)? as usize;
34    let instructions = try_disassemble(
35        &obj.data()[start_offset as usize..call_instr_offset],
36        code_registration,
37    )?;
38    let regs = analyze_reg_rel(obj, &instructions)?;
39
40    Ok(if obj.format() == object::BinaryFormat::Pe {
41        // Microsoft x64 calling convention
42        (regs[&Register::RCX], regs[&Register::RDX])
43    } else {
44        // System V AMD64 ABI
45        (regs[&Register::RDI], regs[&Register::RSI])
46    })
47}

49/// Simple static register analysis for RIP-relative memory loads.
50/// Tracks the latest known values of registers in straight-line code.
51///
52/// # Arguments
53/// * `pe` - The PE file
54/// * `instructions` - Slice of Instructions to analyze
55pub fn analyze_reg_rel(
56    obj: &ObjectReader,
57    instructions: &[Instruction],
58) -> loader::Result<HashMap<Register, u64>> {
59    let mut registry: HashMap<Register, u64> = HashMap::new();
60
61    for instr in instructions {
62        match instr.code() {
63            // lea reg, [disp]
64            iced_x86::Code::Lea_r64_m => {
65                let reg = instr.op0_register();
66                let addr = instr.memory_displacement64();
67                registry.insert(reg, addr);
68            }
69
70            // mov reg, [disp]
71            iced_x86::Code::Mov_r64_rm64 => {
72                let reg = instr.op0_register();
73                if let Some(addr) = mem_operand_target(instr, 1, &registry)
74                    && let Ok(val) = obj.read_u64(addr)
75                {
76                    registry.insert(reg, val);
77                }
78            }
79
80            _ => {}
81        }
82    }
83
84    Ok(registry)
85}

87fn mem_operand_target(
88    instr: &Instruction,
89    operand: u32,
90    regs: &HashMap<Register, u64>,
91) -> Option<u64> {
92    instr.virtual_address(operand, 0, |reg, _, _| {
93        match reg {
94            // The base address of ES, CS, SS and DS is always 0 in 64-bit mode
95            Register::ES | Register::CS | Register::SS | Register::DS => Some(0),
96            _ => regs.get(&reg).cloned(),
97        }
98    })
99}

101fn try_disassemble(code: &[u8], start_addr: u64) -> loader::Result<Vec<Instruction>> {
102    let decoder = Decoder::with_ip(64, code, start_addr, DecoderOptions::NONE);
103
104    Ok(decoder.into_iter().collect::<Vec<Instruction>>())
105}

107fn nth_indirect_call(
108    obj: &ObjectReader,
109    start_vaddr: u64,
110    n: usize,
111) -> loader::Result<Option<u64>> {
112    let start_offset = obj.vaddr_conv(start_vaddr)?;
113    let decoder = Decoder::with_ip(
114        64,
115        &obj.data()[start_offset as usize..],
116        start_vaddr,
117        DecoderOptions::NONE,
118    );
119
120    let Some(indirect_call) = decoder
121        .into_iter()
122        .filter(|t| t.is_call_near_indirect())
123        .nth(n - 1)
124    else {
125        return Ok(None);
126    };
127
128    // ---- Resolve the indirect call target ----
129    let target = match indirect_call.op0_kind() {
130        OpKind::Memory => {
131            // Determine reg values that may be used as part of memory operand
132            let call_offset = obj.vaddr_conv(indirect_call.ip())?;
133            let instructions = try_disassemble(
134                &obj.data()[start_offset as usize..call_offset as usize],
135                start_vaddr,
136            )?;
137            let regs = analyze_reg_rel(obj, &instructions)?;
138
139            let addr = mem_operand_target(&indirect_call, 0, &regs).ok_or_else(|| {
140                Il2CppBinaryError::BadInstruction(indirect_call.to_string(), indirect_call.ip())
141            })?;
142
143            obj.read_u64(addr)?
144        }
145
146        _ => {
147            return Err(Il2CppBinaryError::BadInstruction(
148                "Unsupported indirect call operand".to_string(),
149                indirect_call.ip(),
150            ));
151        }
152    };
153
154    Ok(Some(target))
155}

157/// Find the nth call instruction starting from addr. n=1 is the first call.
158/// Returns the target address of the call
159fn nth_call(obj: &ObjectReader, addr: u64, n: usize) -> loader::Result<u64> {
160    let decoder = Decoder::with_ip(
161        64,
162        &obj.data()[obj.vaddr_conv(addr)? as usize..],
163        addr,
164        DecoderOptions::NONE,
165    );
166    let call_instr = decoder
167        .into_iter()
168        .filter(|t| t.is_call_near())
169        .nth(n - 1)
170        .ok_or_else(|| {
171            Il2CppBinaryError::BadInstruction("Could not find nth call".to_string(), addr)
172        })?;
173
174    Ok(call_instr.near_branch_target())
175}

177/// Returns the address of the nth instruction that matches the predicate
178fn nth_matching(
179    obj: &ObjectReader,
180    addr: u64,
181    n: usize,
182    predicate: impl Fn(&Instruction) -> bool,
183) -> loader::Result<u64> {
184    let decoder = Decoder::with_ip(
185        64,
186        &obj.data()[obj.vaddr_conv(addr)? as usize..],
187        addr,
188        DecoderOptions::NONE,
189    );
190
191    let instr = decoder
192        .into_iter()
193        .filter(|t| predicate(t))
194        .nth(n - 1)
195        .ok_or_else(|| {
196            Il2CppBinaryError::BadInstruction(
197                "Could not find nth matching instruction".to_string(),
198                addr,
199            )
200        })?;
201
202    Ok(instr.ip())
