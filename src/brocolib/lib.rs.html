1//! This library provides a convenient way to parse the metadata structures for
2//! a Unity IL2CPP game.
3//!
4//! The documentation contains many references to C/C++ header source files.
5//! You can find these files in a Unity install at the following path:
6//! `UnityEditor/2021.3.16f1/Editor/Data/il2cpp/libil2cpp`
7
8pub mod global_metadata;
9pub mod runtime_metadata;
10
11use global_metadata::{GlobalMetadata, MetadataDeserializeError};
12use runtime_metadata::RuntimeMetadata;
13use thiserror::Error;
14
15use crate::runtime_metadata::loader::Il2CppBinaryError;
16
17/// A container for all of the applications metadata structures.
18///
19/// A Unity IL2CPP application stores metadata in two different ways, the
20/// global metadata and the runtime metadata.
21///
22/// The global metadata is generally the `global-metadata.dat` file in the
23/// application. See [`GlobalMetadata`] for more information.
24///
25/// The runtime metadata is stored inside the game binary itself. This is
26/// generally the `libil2cpp.so` file in the application. See
27/// [`RuntimeMetadata`] for more information.
28pub struct Metadata<'gmd, 'rmd> {
29    /// The application's global metadata.
30    ///
31    /// See [`GlobalMetadata`] for more information.
32    pub global_metadata: GlobalMetadata<'gmd>,
33
34    /// The application's runtime metadata.
35    ///
36    /// See [`RuntimeMetadata`] for more information.
37    pub runtime_metadata: RuntimeMetadata<'rmd>,
38}
39
40#[derive(Error, Debug)]
41pub enum MetadataParseError {
42    #[error("could not parse global metadata")]
43    GlobalMetadata(#[from] MetadataDeserializeError),
44
45    #[error("could not parse runtime metadata")]
46    Binary(#[from] Il2CppBinaryError),
47}
48
49impl<'gmd, 'rmd> Metadata<'gmd, 'rmd> {
50    pub fn parse(global_metadata: &'gmd [u8], lib: &'rmd [u8]) -> Result<Self, MetadataParseError> {
51        let global_metadata = global_metadata::deserialize(global_metadata)?;
52        let runtime_metadata = RuntimeMetadata::read_obj(lib, &global_metadata)?;
53
54        Ok(Metadata {
55            global_metadata,
56            runtime_metadata,
57        })
58    }
59}
